-- VERIFICAR CARACAS
SELECT 
    'CARACAS' AS tienda,
    id,
    nro_factura,
    nombre_apellido,
    total_depositado_usd AS "Depositado USD",
    deuda_usd AS "Deuda USD",
    monto_depositados AS "Depositado Bs",
    deuda AS "Deuda Bs",
    cuotas_pagadas AS "Cuotas Pagadas",
    (SELECT COUNT(*) FROM pagos_caracas p WHERE p.factura_id = t.id) AS "Pagos"
FROM tienda_caracas t
ORDER BY id;
